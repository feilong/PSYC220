Sampling Distributions

Draw many samples of the same size and watch each statistic form its own distribution

Set a sample size, then draw sample after sample from the same population. Each sample gives you one mean, one SS, one variance, one SD — and those values pile up into a distribution of their own. That pile is the sampling distribution.

from shiny import App, render, ui, reactive
import numpy as np
import matplotlib.pyplot as plt

C_BAR = "darkorange"    # every histogram on this page
C_OBS = "purple"        # the observed average of whatever the panel shows
C_DATA = "dodgerblue"
C_TRUE = "black"

app_ui = ui.page_fluid(
    ui.layout_sidebar(
        ui.sidebar(
            ui.input_select(
                "dist", "Population shape",
                {"norm": "Normal (symmetric, bell)",
                 "unif": "Uniform (flat)",
                 "exp": "Exponential (right-skewed)",
                 "bimodal": "Bimodal (two peaks)",
                 "beta": "Beta (left-skewed)"}
            ),
            ui.input_numeric("pop_mu", "Population mean (μ):", value=100.0, step=1.0),
            ui.input_numeric("pop_var", "Population variance (σ²):", value=225.0,
                             min=0.1, step=5.0),
            ui.input_numeric("n", "Sample size (n):", value=10, min=2, step=1),
            ui.hr(),
            ui.input_action_button("draw_1", "Draw 1 sample", class_="btn-primary"),
            ui.input_action_button("draw_10", "Draw 10", class_="btn-success"),
            ui.input_action_button("draw_100", "Draw 100", class_="btn-warning"),
            ui.input_action_button("reset", "Clear", class_="btn-danger"),
            ui.hr(),
            ui.output_ui("readout"),
            width=300
        ),
        ui.output_plot("main_plot", height="1000px")
    )
)


def server(input, output, session):
    hist = reactive.Value({"means": [], "ss": [], "var": [], "sd": []})

    def base_params(d):
        return {"norm": (0.0, 1.0),
                "unif": (0.5, np.sqrt(1 / 12)),
                "exp": (1.0, 1.0),
                "bimodal": (0.0, np.sqrt(4.25)),
                "beta": (0.8, np.sqrt(16 / 1100))}[d]

    def draw_samples(k):
        d = input.dist()
        mu, sigma = float(input.pop_mu()), np.sqrt(float(input.pop_var()))
        n = int(input.n())
        b_mu, b_sigma = base_params(d)
        size = (k, n)
        if d == "norm":
            raw = np.random.standard_normal(size)
        elif d == "unif":
            raw = np.random.uniform(size=size)
        elif d == "exp":
            raw = np.random.exponential(size=size)
        elif d == "bimodal":
            raw = np.random.normal(np.where(np.random.rand(k, n) < 0.5, -2, 2), 0.5)
        else:
            raw = np.random.beta(8, 2, size=size)
        x = (raw - b_mu) * (sigma / b_sigma) + mu

        means = x.mean(axis=1)
        ss = np.sum((x - means[:, None]) ** 2, axis=1)
        var = ss / (n - 1)
        sd = np.sqrt(var)

        cur = hist.get().copy()
        cur["means"] = cur["means"] + means.tolist()
        cur["ss"] = cur["ss"] + ss.tolist()
        cur["var"] = cur["var"] + var.tolist()
        cur["sd"] = cur["sd"] + sd.tolist()
        hist.set(cur)

    @reactive.Effect
    @reactive.event(input.draw_1)
    def _d1():
        draw_samples(1)

    @reactive.Effect
    @reactive.event(input.draw_10)
    def _d10():
        draw_samples(10)

    @reactive.Effect
    @reactive.event(input.draw_100)
    def _d100():
        draw_samples(100)

    @reactive.Effect
    @reactive.event(input.reset, input.dist, input.pop_mu, input.pop_var, input.n)
    def _clear():
        hist.set({"means": [], "ss": [], "var": [], "sd": []})

    def pop_curve(d, mu, sigma):
        x = np.linspace(mu - 4 * sigma, mu + 4 * sigma, 800)
        b_mu, b_sigma = base_params(d)
        scale = sigma / b_sigma
        xb = (x - mu) / scale + b_mu
        if d == "norm":
            yb = np.exp(-0.5 * xb ** 2) / np.sqrt(2 * np.pi)
        elif d == "unif":
            yb = np.where((xb >= 0) & (xb <= 1), 1.0, 0.0)
        elif d == "exp":
            yb = np.where(xb >= 0, np.exp(-xb), 0.0)
        elif d == "bimodal":
            yb = (np.exp(-2 * (xb + 2) ** 2) + np.exp(-2 * (xb - 2) ** 2)) / np.sqrt(2 * np.pi)
        else:
            yb = np.where((xb >= 0) & (xb <= 1), 72 * xb ** 7 * (1 - xb), 0.0)
        return x, yb / scale

    @output
    @render.ui
    def readout():
        h = hist.get()
        k = len(h["means"])
        if k == 0:
            return ui.markdown("*No samples yet.*")
        n = int(input.n())
        sigma = np.sqrt(float(input.pop_var()))
        return ui.markdown(
            f"**{k} samples of n = {n}**  \n"
            f"mean of M = {np.mean(h['means']):.2f}  \n"
            f"mean of SS = {np.mean(h['ss']):.1f}  \n"
            f"mean of s² = {np.mean(h['var']):.2f}  \n"
            f"mean of s = {np.mean(h['sd']):.2f}  (σ = {sigma:.2f})")

    @output
    @render.plot
    def main_plot():
        h = hist.get()
        k = len(h["means"])
        d = input.dist()
        mu = float(input.pop_mu())
        true_var = float(input.pop_var())
        sigma = np.sqrt(true_var)
        n = int(input.n())

        fig = plt.figure(figsize=(12, 11))
        gs = fig.add_gridspec(3, 2, height_ratios=[0.9, 1, 1], hspace=0.42, wspace=0.26)
        ax_pop = fig.add_subplot(gs[0, :])
        ax_mean = fig.add_subplot(gs[1, 0])
        ax_ss = fig.add_subplot(gs[1, 1])
        ax_var = fig.add_subplot(gs[2, 0])
        ax_sd = fig.add_subplot(gs[2, 1])

        x, y = pop_curve(d, mu, sigma)
        ax_pop.plot(x, y, color="gray", lw=2)
        ax_pop.fill_between(x, y, alpha=0.2, color="gray")
        ax_pop.axvline(mu, color=C_TRUE, ls="--", lw=2, label=f"μ = {mu:g}")
        ax_pop.set_xlim(mu - 4 * sigma, mu + 4 * sigma)
        ax_pop.set_title(f"Population — every sample of n = {n} comes from here")
        ax_pop.legend(loc="upper right")

        panels = [(ax_mean, "Sampling distribution of M"),
                  (ax_ss, "Sampling distribution of SS"),
                  (ax_var, "Sampling distribution of s²"),
                  (ax_sd, "Sampling distribution of s")]
        if k == 0:
            for ax, title in panels:
                ax.set_title(title)
                ax.text(0.5, 0.5, "Draw some samples", ha="center", va="center",
                        transform=ax.transAxes, fontsize=12, color="gray")
                ax.set_xticks([]); ax.set_yticks([])
            return fig

        bins = min(30, max(8, int(np.sqrt(k))))

        # mean: centres on mu, spread is the standard error
        ax_mean.hist(h["means"], bins=bins, density=True, color=C_BAR,
                     edgecolor="white", alpha=0.75)
        se = sigma / np.sqrt(n)
        xm = np.linspace(mu - 4 * se, mu + 4 * se, 300)
        ax_mean.plot(xm, np.exp(-0.5 * ((xm - mu) / se) ** 2) / (se * np.sqrt(2 * np.pi)),
                     color="seagreen", lw=2, label=r"$N(\mu,\ \sigma^2/n)$")
        ax_mean.axvline(mu, color=C_TRUE, ls="--", lw=2, label=f"μ = {mu:g}")
        ax_mean.axvline(np.mean(h["means"]), color=C_OBS, lw=2,
                        label=f"mean of M = {np.mean(h['means']):.2f}")
        ax_mean.set_title(f"M over {k} samples — centres on μ, SE = {se:.2f}")
        ax_mean.set_xlabel("sample mean M")
        ax_mean.legend(loc="upper right")

        # SS: expectation is (n-1) sigma^2
        ax_ss.hist(h["ss"], bins=bins, color=C_BAR, edgecolor="white", alpha=0.75)
        exp_ss = (n - 1) * true_var
        ax_ss.axvline(exp_ss, color=C_TRUE, ls="--", lw=2,
                      label=f"$(n-1)\\sigma^2$ = {exp_ss:.0f}")
        ax_ss.axvline(np.mean(h["ss"]), color=C_OBS, lw=2,
                      label=f"mean of SS = {np.mean(h['ss']):.0f}")
        ax_ss.set_title("SS — depends on n, so it has no fixed target")
        ax_ss.set_xlabel(r"$\sum (X-M)^2$")
        ax_ss.legend(loc="upper right")

        # variance: unbiased, averages to sigma^2
        ax_var.hist(h["var"], bins=bins, color=C_BAR, edgecolor="white", alpha=0.75)
        ax_var.axvline(true_var, color=C_TRUE, ls="--", lw=2, label=f"σ² = {true_var:g}")
        ax_var.axvline(np.mean(h["var"]), color=C_OBS, lw=2,
                       label=f"mean of s² = {np.mean(h['var']):.1f}")
        ax_var.set_title("s² is unbiased — its average lands on σ²")
        ax_var.set_xlabel(r"$s^2$")
        ax_var.legend(loc="upper right")

        # sd: NOT unbiased, averages a little below sigma
        ax_sd.hist(h["sd"], bins=bins, color=C_BAR, edgecolor="white", alpha=0.75)
        ax_sd.axvline(sigma, color=C_TRUE, ls="--", lw=2, label=f"σ = {sigma:.2f}")
        ax_sd.axvline(np.mean(h["sd"]), color=C_OBS, lw=2,
                      label=f"mean of s = {np.mean(h['sd']):.2f}")
        ax_sd.set_title("s sits slightly below σ on average")
        ax_sd.set_xlabel(r"$s$")
        ax_sd.legend(loc="upper right")

        return fig


app = App(app_ui, server)

What to look for

  • M piles up into a bell curve centred on \(\mu\), whatever shape the population has — that is the Central Limit Theorem. Its spread is the standard error, \(\sigma/\sqrt{n}\), so raise \(n\) and the pile narrows.
  • SS has no fixed target. Change \(n\) and the whole distribution slides, because SS is a total. Its average sits at \((n-1)\sigma^2\).
  • s² is unbiased: draw enough samples and the average of all your \(s^2\) values lands on \(\sigma^2\). This is what dividing by \(n-1\) buys you.
  • s is the interesting one. Even though \(s^2\) averages to \(\sigma^2\), the average of \(s\) lands slightly below \(\sigma\) — taking a square root does not preserve the unbiasedness. Try a small \(n\), like 2 or 3, and the gap is easy to see.
