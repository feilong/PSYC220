Growing a Sample

How the mean, SS, variance, and standard deviation change as n grows

Draw scores one at a time from a population you choose, and watch four statistics update as the sample grows. The population values you set are the truth; the sample statistics are estimates of them, and the point of the demo is to see how each one behaves as \(n\) increases.

from shiny import App, render, ui, reactive
import numpy as np
import matplotlib.pyplot as plt

C_MU = "purple"        # anything computed about the population mean, mu
C_M = "darkorange"     # anything computed about the sample mean, M
C_DATA = "dodgerblue"
C_TRUE = "black"

app_ui = ui.page_fluid(
    ui.layout_sidebar(
        ui.sidebar(
            ui.input_select(
                "dist", "Population shape",
                {"norm": "Normal (symmetric, bell)",
                 "unif": "Uniform (flat)",
                 "exp": "Exponential (right-skewed)",
                 "bimodal": "Bimodal (two peaks)",
                 "beta": "Beta (left-skewed)"}
            ),
            ui.input_numeric("pop_mu", "Population mean (μ):", value=100.0, step=1.0),
            ui.input_numeric("pop_var", "Population variance (σ²):", value=225.0,
                             min=0.1, step=5.0),
            ui.hr(),
            ui.input_action_button("draw_1", "Add 1 score", class_="btn-primary"),
            ui.input_action_button("draw_10", "Add 10 scores", class_="btn-success"),
            ui.input_action_button("reset", "Start over", class_="btn-danger"),
            ui.hr(),
            ui.input_checkbox("show_m", "Deviations about M (sample mean)", value=True),
            ui.input_checkbox("show_mu", "Deviations about μ (population mean)", value=False),
            ui.hr(),
            ui.output_ui("readout"),
            width=300
        ),
        ui.output_plot("main_plot", height="1000px")
    )
)


def server(input, output, session):
    sample_data = reactive.Value([])

    # mean and sd of each raw distribution, before it is shifted and scaled
    def base_params(d):
        return {"norm": (0.0, 1.0),
                "unif": (0.5, np.sqrt(1 / 12)),
                "exp": (1.0, 1.0),
                "bimodal": (0.0, np.sqrt(4.25)),
                "beta": (0.8, np.sqrt(16 / 1100))}[d]

    def draw(k):
        d = input.dist()
        mu, sigma = float(input.pop_mu()), np.sqrt(float(input.pop_var()))
        b_mu, b_sigma = base_params(d)
        if d == "norm":
            raw = np.random.standard_normal(k)
        elif d == "unif":
            raw = np.random.uniform(size=k)
        elif d == "exp":
            raw = np.random.exponential(size=k)
        elif d == "bimodal":
            raw = np.random.normal(np.where(np.random.rand(k) < 0.5, -2, 2), 0.5)
        else:
            raw = np.random.beta(8, 2, size=k)
        return ((raw - b_mu) * (sigma / b_sigma) + mu).tolist()

    @reactive.Effect
    @reactive.event(input.draw_1)
    def _add1():
        sample_data.set(sample_data() + draw(1))

    @reactive.Effect
    @reactive.event(input.draw_10)
    def _add10():
        sample_data.set(sample_data() + draw(10))

    @reactive.Effect
    @reactive.event(input.reset, input.dist, input.pop_mu, input.pop_var)
    def _clear():
        sample_data.set([])

    def pop_curve(d, mu, sigma):
        x = np.linspace(mu - 4 * sigma, mu + 4 * sigma, 800)
        b_mu, b_sigma = base_params(d)
        scale = sigma / b_sigma
        xb = (x - mu) / scale + b_mu
        if d == "norm":
            yb = np.exp(-0.5 * xb ** 2) / np.sqrt(2 * np.pi)
        elif d == "unif":
            yb = np.where((xb >= 0) & (xb <= 1), 1.0, 0.0)
        elif d == "exp":
            yb = np.where(xb >= 0, np.exp(-xb), 0.0)
        elif d == "bimodal":
            yb = (np.exp(-2 * (xb + 2) ** 2) + np.exp(-2 * (xb - 2) ** 2)) / np.sqrt(2 * np.pi)
        else:
            yb = np.where((xb >= 0) & (xb <= 1), 72 * xb ** 7 * (1 - xb), 0.0)
        return x, yb / scale

    @output
    @render.ui
    def readout():
        data = np.array(sample_data())
        n = len(data)
        if n == 0:
            return ui.markdown("*No scores yet.*")
        m = data.mean()
        ss = float(np.sum((data - m) ** 2))
        if n > 1:
            var = ss / (n - 1)
            return ui.markdown(
                f"**n = {n}**  \nM = {m:.2f}  \nSS = {ss:.1f}  \n"
                f"s² = {var:.2f}  \ns = {np.sqrt(var):.2f}")
        ss_mu = float((data[0] - float(input.pop_mu())) ** 2)
        return ui.markdown(
            f"**n = 1**  \nM = {m:.2f}  \n"
            f"SS about M = 0  \nSS about μ = {ss_mu:.1f}  \n"
            f"*s² and s need n ≥ 2*")

    @output
    @render.plot
    def main_plot():
        data = np.array(sample_data())
        n = len(data)
        d = input.dist()
        mu = float(input.pop_mu())
        true_var = float(input.pop_var())
        sigma = np.sqrt(true_var)
        show_mu = bool(input.show_mu())
        show_m = bool(input.show_m())

        fig = plt.figure(figsize=(12, 11))
        gs = fig.add_gridspec(3, 2, height_ratios=[0.9, 1, 1], hspace=0.42, wspace=0.26)
        ax_pop = fig.add_subplot(gs[0, :])
        ax_mean = fig.add_subplot(gs[1, 0])
        ax_ss = fig.add_subplot(gs[1, 1])
        ax_var = fig.add_subplot(gs[2, 0])
        ax_sd = fig.add_subplot(gs[2, 1])

        x, y = pop_curve(d, mu, sigma)
        ax_pop.plot(x, y, color="gray", lw=2)
        ax_pop.fill_between(x, y, alpha=0.2, color="gray")
        ax_pop.axvline(mu, color=C_TRUE, ls="--", lw=2, label=f"μ = {mu:g}")
        ax_pop.set_xlim(mu - 4 * sigma, mu + 4 * sigma)
        ax_pop.set_title("Population, and the scores drawn so far")
        if n:
            ax_pop.scatter(data, np.zeros(n), color=C_DATA, zorder=5, alpha=0.6, label="scores")
            ax_pop.axvline(data.mean(), color=C_M, lw=2, label=f"M = {data.mean():.2f}")
        ax_pop.legend(loc="upper right")

        panels = [(ax_mean, "Sample mean M"), (ax_ss, "Sum of squares SS"),
                  (ax_var, "Variance"), (ax_sd, "Standard deviation")]
        if n == 0:
            for ax, title in panels:
                ax.set_title(title)
                ax.text(0.5, 0.5, "Add a score to begin", ha="center", va="center",
                        transform=ax.transAxes, fontsize=12, color="gray")
                ax.set_xticks([]); ax.set_yticks([])
            return fig

        ns = np.arange(1, n + 1)
        means, ss_mu, ss_m, var_n1, var_n, sd_n1, sd_n = [], [], [], [], [], [], []
        for i in ns:
            sub = data[:i]
            m = sub.mean()
            means.append(m)
            ss_mu.append(np.sum((sub - mu) ** 2))
            s_m = np.sum((sub - m) ** 2)
            ss_m.append(s_m)
            var_n.append(np.sum((sub - mu) ** 2) / i)
            var_n1.append(s_m / (i - 1) if i > 1 else np.nan)
            sd_n.append(np.sqrt(var_n[-1]))
            sd_n1.append(np.sqrt(var_n1[-1]) if i > 1 else np.nan)

        # mean
        ax_mean.axhline(mu, color=C_TRUE, ls="--", lw=2, label=f"μ = {mu:g}")
        ax_mean.plot(ns, means, color=C_M, marker="o", ms=3, label="M")
        ax_mean.set_title("Sample mean settles on μ")
        ax_mean.set_xlabel("n"); ax_mean.set_ylabel("M")
        ax_mean.legend(loc="upper right")

        # SS -- the one that grows without bound
        if show_mu:
            ax_ss.plot(ns, ss_mu, color=C_MU, marker="o", ms=3, label=r"$\sum (X-\mu)^2$")
        if show_m:
            ax_ss.plot(ns, ss_m, color=C_M, marker="s", ms=3, label=r"$\sum (X-M)^2$")
        ax_ss.set_title("SS keeps growing — it is a total")
        ax_ss.set_xlabel("n"); ax_ss.set_ylabel("SS")
        if show_mu or show_m:
            ax_ss.legend(loc="upper left")

        # variance
        ax_var.axhline(true_var, color=C_TRUE, ls="--", lw=2, label=f"σ² = {true_var:g}")
        shown = []
        if show_mu:
            ax_var.plot(ns, var_n, color=C_MU, marker="o", ms=3, alpha=0.8,
                        label=r"$\sum (X-\mu)^2 / n$")
            shown += list(var_n)
        if show_m and n > 1:
            ax_var.plot(ns, var_n1, color=C_M, marker="s", ms=3, alpha=0.8,
                        label=r"$\sum (X-M)^2 / (n-1)$")
            shown += list(var_n1)
        elif show_m:
            ax_var.text(0.5, 0.08, "needs n ≥ 2", ha="center", color="gray",
                        transform=ax_var.transAxes, fontsize=10)
        ax_var.set_title("Variance is an average — it settles")
        ax_var.set_xlabel("n"); ax_var.set_ylabel("variance")
        vals = [v for v in shown if np.isfinite(v)] + [true_var]
        ax_var.set_ylim(0, max(vals) * 1.35)
        ax_var.legend(loc="upper right")

        # sd
        ax_sd.axhline(sigma, color=C_TRUE, ls="--", lw=2, label=f"σ = {sigma:.2f}")
        shown = []
        if show_mu:
            ax_sd.plot(ns, sd_n, color=C_MU, marker="o", ms=3, alpha=0.8,
                       label=r"$\sqrt{\sum (X-\mu)^2 / n}$")
            shown += list(sd_n)
        if show_m and n > 1:
            ax_sd.plot(ns, sd_n1, color=C_M, marker="s", ms=3, alpha=0.8,
                       label=r"$\sqrt{\sum (X-M)^2 / (n-1)}$")
            shown += list(sd_n1)
        elif show_m:
            ax_sd.text(0.5, 0.08, "needs n ≥ 2", ha="center", color="gray",
                       transform=ax_sd.transAxes, fontsize=10)
        ax_sd.set_title("SD is the variance back in the original units")
        ax_sd.set_xlabel("n"); ax_sd.set_ylabel("SD")
        vals = [v for v in shown if np.isfinite(v)] + [sigma]
        ax_sd.set_ylim(0, max(vals) * 1.35)
        ax_sd.legend(loc="upper right")

        return fig


app = App(app_ui, server)

What to look for

  • M wanders at first and then settles on \(\mu\). With a handful of scores it can be well off.
  • SS never settles — it climbs with every score you add, because it is a total, not an average. That is why SS on its own cannot tell you whether one group is more spread out than another.
  • Variance does settle, because dividing by \(n\) (or \(n-1\)) turns the total into an average.
  • SD settles too, and in the same units as the original scores — which is why it is the one you report.

The purple and orange curves differ only in which mean the deviations are taken from. The orange one is always the smaller of the two, because \(M\) is the value that makes \(\sum (X-c)^2\) as small as it can be.
